Copy column I into each YouTube description.'; } // ============================================================ // LOG ONE ANONYMOUS CLICK (called from markClicked when no email) // ============================================================ function logYouTubeClick_(utmCampaign, utmContent, utmSource, utmMedium, landing, videoIdHint) { try { var ss = SpreadsheetApp.openById(SHEET_ID); var ls = ss.getSheetByName(YT_CLICK_SHEET); if (!ls) { setupYouTubeSheets(); ls = ss.getSheetByName(YT_CLICK_SHEET); } var videoId = videoIdHint || lookupVideoId_(utmCampaign, utmContent) || ''; ls.appendRow([ new Date(), videoId, utmCampaign || '', utmContent || '', utmSource || 'youtube', utmMedium || 'video', landing || YT_LANDING_PAGE ]); // Also append a lightweight row to Campaign_Log so dashboard stays unified try { var cl = ss.getSheetByName('Campaign_Log'); if (cl) { var clickId = 'YTCLK_' + Utilities.getUuid().substring(0, 8); cl.appendRow([ clickId, // A CampaignID (click id) 'YOUTUBE_CLICK', // B ContactID '', // C Email (unknown yet) utmContent || '', // D Industry / content tag '', // E Version 'YouTube click: ' + (utmCampaign || ''), // F Subject new Date(), // G SentDate / ClickTime 'YES', // H Clicked landing || YT_LANDING_PAGE, // I CTA_Link 'YouTube', // J Campaign name '', // K Dashboard key utmCampaign || '' // L Campaign_Code ]); } } catch (e2) { Logger.log('[YouTube→Campaign_Log] ' + e2.message); } } catch (err) { Logger.log('[YouTube Click Log Error] ' + err.message); } } function lookupVideoId_(utmCampaign, utmContent) { try { var sheet = SpreadsheetApp.openById(SHEET_ID).getSheetByName(YT_CAMPAIGN_SHEET); if (!sheet) return ''; var data = sheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { var camp = data[i][2] ? data[i][2].toString().trim() : ''; var cont = data[i][3] ? data[i][3].toString().trim() : ''; if (utmCampaign && camp === utmCampaign) return data[i][0].toString(); if (utmContent && cont === utmContent) return data[i][0].toString(); } } catch (e) {} return ''; } // ============================================================ // REFRESH CLICK COUNTS into YouTube_Campaigns column J // ============================================================ function refreshYouTubeClickCounts() { var ss = SpreadsheetApp.openById(SHEET_ID); var ls = ss.getSheetByName(YT_CLICK_SHEET); var cs = ss.getSheetByName(YT_CAMPAIGN_SHEET); if (!ls || !cs) return 'Sheets missing — run Setup first.'; var clicks = ls.getDataRange().getValues(); var byCampaign = {}; var byContent = {}; for (var i = 1; i < clicks.length; i++) { var camp = clicks[i][2] ? clicks[i][2].toString().trim() : ''; var cont = clicks[i][3] ? clicks[i][3].toString().trim() : ''; if (camp) byCampaign[camp] = (byCampaign[camp] || 0) + 1; if (cont) byContent[cont] = (byContent[cont] || 0) + 1; } var camp = cs.getDataRange().getValues(); var out = []; for (var r = 1; r < camp.length; r++) { var cCode = camp[r][2] ? camp[r][2].toString().trim() : ''; var cCont = camp[r][3] ? camp[r][3].toString().trim() : ''; var n = byCampaign[cCode] || byContent[cCont] || 0; out.push([n]); } if (out.length > 0) cs.getRange(2, 10, out.length, 1).setValues(out); Logger.log('[YouTube] Click counts refreshed.'); return 'Click counts refreshed for ' + out.length + ' videos.'; }
