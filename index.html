// ============================================================ // 6) getCampaignByIdAnyStatus_ — ADD (for Test Send attachments) // ============================================================ function getCampaignByIdAnyStatus_(campaignId) { var data = SpreadsheetApp.openById(SHEET_ID) .getSheetByName('Campaign_Config').getDataRange().getValues(); for (var i = 1; i < data.length; i++) { var row = data[i]; var id = row[0] ? row[0].toString().trim() : ''; if (id !== campaignId) continue; var campaignCode = (row.length > 11 && row[11] && row[11].toString().trim() !== '') ? row[11].toString().trim() : ('NL-' + id.replace(/[^a-zA-Z0-9]/g, '').toUpperCase()); return { campaignId: id, campaignName: row[1] ? row[1].toString().trim() : id, campaignCode: campaignCode, knowledgeDocId: row[3] ? row[3].toString().trim() : '', defaultRedirect: row[4] ? row[4].toString().trim() : DEFAULT_REDIRECT, dailyBatch: row[5] ? parseInt(row[5]) : DAILY_TOTAL_BATCH, cooldownDays: row[6] ? parseInt(row[6]) : COOLDOWN_DAYS, senderName: row[7] ? row[7].toString().trim() : 'BioEnable Team', ccEmail: row[8] ? row[8].toString().trim() : '', ctaLabel: (row.length > 9 && row[9]) ? row[9].toString().trim() : '', attachmentId: (row.length > 10 && row[10]) ? row[10].toString().trim() : '' }; } return null; }
